    match /settings/main {

    match /listings/{listingId} {
      allow read: if true;
      allow create: if request.auth != null && request.auth.token.status == 'active';
      allow update, delete: if request.auth != null && (
        request.auth.token.role == 'owner' ||
        request.auth.token.role == 'moderator' ||
        resource.data.ownerId == request.auth.uid
      );
    }
    match /users/{userId} {
      allow read, write: if request.auth != null && (
        request.auth.uid == userId || request.auth.token.role == 'owner'
      );
    }
    match /auditLogs/{logId} {
      allow read: if request.auth != null && request.auth.token.role == 'owner';
      allow create: if request.auth != null;
    }
  }
